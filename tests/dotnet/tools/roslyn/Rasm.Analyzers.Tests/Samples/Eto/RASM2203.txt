using System;
using Eto.Forms;
using LanguageExt;
using Rasm.Rhino;
using static LanguageExt.Prelude;

namespace Rasm.Rhino {
    public sealed class Disposal<T>(T held, Action<T> release) : IDisposable {
        public void Dispose() => release(held);
    }

    public sealed class Retained<T> : IDisposable where T : IDisposable {
        public void Retain(T value) => value.Dispose();
        public void Dispose() { }
    }

    public static class DisposalOps {
        public static IO<T> Acquire<T, TLease>(IO<T> acquire, Func<T, TLease> lease, Action<TLease> retain) where TLease : IDisposable =>
            from held in acquire
            let owned = lease(held)
            from retained in OnFailure(IO.lift(() => retain(owned)), IO.lift(owned.Dispose))
            select held;
        public static IO<T> Acquire<T>(IO<T> acquire, Action<T> retain) where T : IDisposable => Acquire(acquire, static held => held, retain);
        public static IO<T> OnFailure<T>(IO<T> effect, IO<Unit> release) => effect | release.Bind(_ => effect);
        public static IO<Unit> Release<T>(params T[] held) where T : IDisposable => IO.lift(() => System.Array.ForEach(held, static item => item.Dispose()));
    }
}

internal sealed class RowGrid(Retained<IDisposable> acquired) : Button {
    public Retained<IDisposable> Acquired { get; } = acquired;
    public IO<Unit> Attached => IO.pure(unit);
}

internal static class RetainedThenDisposed {
    internal static IO<RowGrid> Realize(Action<IDisposable> retain) =>
        from acquired in IO.lift(static () => new Retained<IDisposable>())
        from grid in DisposalOps.OnFailure(
            from created in DisposalOps.Acquire(IO.lift(() => new RowGrid(acquired)), retain)
            select created,
            IO.lift(acquired.Dispose))
        from attached in DisposalOps.OnFailure(IO.lift(() => grid.Style = "grid").Bind(_ => grid.Attached), IO.lift({|#0:grid.Dispose|}))
        select grid;

    internal static IO<Unit> Shown(Action<IDisposable> retain) =>
        from panel in DisposalOps.Acquire(IO.lift(static () => new Button()), retain)
        from shown in DisposalOps.OnFailure(Show(panel), IO.lift(() => {|#1:panel.Dispose()|}))
        select shown;

    internal static IO<Unit> Kept(Action<IDisposable> retain) =>
        from panel in IO.lift(static () => new Button())
        from kept in IO.lift(() => retain(panel))
        from shown in DisposalOps.OnFailure(Show(panel), {|#2:DisposalOps.Release(panel)|})
        select shown;

    internal static IO<Unit> Bound(Action<IDisposable> retain) =>
        DisposalOps.Acquire(IO.lift(static () => new Button()), retain).Bind(panel => DisposalOps.OnFailure(Show(panel), IO.lift({|#3:panel.Dispose|})));

    private static IO<Unit> Show(Button panel) => IO.lift(() => { panel.Visible = true; });
}

internal static class DisposedOnce {
    internal static IO<RowGrid> Realize(Action<IDisposable> retain) =>
        from acquired in IO.lift(static () => new Retained<IDisposable>())
        from grid in DisposalOps.OnFailure(
            from created in DisposalOps.Acquire(IO.lift(() => new RowGrid(acquired)), retain)
            select created,
            IO.lift(acquired.Dispose))
        from attached in IO.lift(() => grid.Style = "grid").Bind(_ => grid.Attached)
        select grid;

    internal static IO<Unit> Opened(Func<IO<Button>> built) =>
        from body in built()
        from opened in DisposalOps.OnFailure(IO.lift(() => { body.Visible = true; }), IO.lift(body.Dispose))
        select opened;

    internal static IO<Unit> Later(Action<IDisposable> retain) =>
        from panel in IO.lift(static () => new Button())
        from shown in DisposalOps.OnFailure(IO.lift(() => { panel.Visible = true; }), IO.lift(panel.Dispose))
        from kept in IO.lift(() => retain(panel))
        select kept;

    internal static IO<Unit> Leased(Action<IDisposable> retain) =>
        from lease in DisposalOps.Acquire(IO.lift(static () => new Disposal<int>(0, static _ => { })), retain)
        from started in DisposalOps.OnFailure(IO.pure(unit), IO.lift(lease.Dispose))
        select started;
}
