using System;
using System.Runtime.CompilerServices;
using Rasm.Rhino.Events;
using Rhino;

namespace Rasm.Rhino.Events {
    public static class Subscriptions {
        public static Type Host<TArgs>(Type owner, Action<EventHandler<TArgs>> add, Action<EventHandler<TArgs>> remove, [CallerMemberName] string member = "") => owner;

        public static Type During<TArgs>(Type owner, Action<EventHandler<TArgs>> add, Action<EventHandler<TArgs>> remove, [CallerMemberName] string member = "") => owner;
    }
}

internal static class Accessors {
    [UnsafeAccessor(UnsafeAccessorKind.StaticMethod, Name = "add_Hidden")]
    internal static extern void AddHidden([UnsafeAccessorType("Rhino.Internal.Settings, RhinoCommon")] object? owner, EventHandler<EventArgs> handler);

    [UnsafeAccessor(UnsafeAccessorKind.StaticMethod, Name = "remove_Hidden")]
    internal static extern void RemoveHidden([UnsafeAccessorType("Rhino.Internal.Settings, RhinoCommon")] object? owner, EventHandler<EventArgs> handler);

    [UnsafeAccessor(UnsafeAccessorKind.StaticMethod, Name = "add_Closing")]
    internal static extern void AddClosing([UnsafeAccessorType("Rhino.RhinoDoc, RhinoCommon")] object? owner, EventHandler<EventArgs> handler);

    [UnsafeAccessor(UnsafeAccessorKind.StaticMethod, Name = "remove_Closing")]
    internal static extern void RemoveClosing([UnsafeAccessorType("Rhino.RhinoDoc, RhinoCommon")] object? owner, EventHandler<EventArgs> handler);
}

internal static class WrongOwners {
    internal static Type Opened => Subscriptions.Host<DocumentOpenEventArgs>({|#0:typeof(Accessors)|}, static h => RhinoDoc.EndOpenDocument += h, static h => RhinoDoc.EndOpenDocument -= h);

    internal static Type Hidden => Subscriptions.During<EventArgs>({|#1:typeof(Accessors)|}, static h => Accessors.AddHidden(owner: null, h), static h => Accessors.RemoveHidden(owner: null, h));
}

internal static class StatedOwners {
    internal static Type Opened => Subscriptions.Host<DocumentOpenEventArgs>(typeof(RhinoDoc), static h => RhinoDoc.EndOpenDocument += h, static h => RhinoDoc.EndOpenDocument -= h);

    internal static Type Closing => Subscriptions.During<EventArgs>(typeof(RhinoDoc), static h => Accessors.AddClosing(owner: null, h), static h => Accessors.RemoveClosing(owner: null, h));
}
