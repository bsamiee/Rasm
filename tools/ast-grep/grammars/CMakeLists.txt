cmake_minimum_required(VERSION 4.4)

# --- [COMPOSITION] ----------------------------------------------------------------------

project(grammar LANGUAGES C)

set(GRAMMAR_SOURCE "" CACHE FILEPATH "Grammar extension or composition")
set(NODE_MODULES "" CACHE PATH "Installed packages")
set(COMPONENTS "" CACHE STRING "External scanners as <scanner.c>|<grammar.json> pairs")
set(SCANNER_COMPILE_OPTIONS "" CACHE STRING "External scanner compiler options")
set(AST_GREP_CONFIG "" CACHE FILEPATH "Project language registration")

file(REAL_PATH "${GRAMMAR_SOURCE}" GRAMMAR_SOURCE)
file(REAL_PATH "${NODE_MODULES}" NODE_MODULES)
cmake_path(GET GRAMMAR_SOURCE PARENT_PATH grammar_dir)
cmake_path(GET CMAKE_CURRENT_BINARY_DIR FILENAME language)
cmake_path(GET CMAKE_CURRENT_BINARY_DIR PARENT_PATH library_dir)
set(parser_dir "${CMAKE_CURRENT_BINARY_DIR}/src")

file(REMOVE_RECURSE "${parser_dir}")
execute_process(COMMAND tree-sitter generate --output "${parser_dir}" "${GRAMMAR_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
file(READ "${parser_dir}/grammar.json" grammar)
string(JSON name GET "${grammar}" name)
string(JSON external_count LENGTH "${grammar}" externals)
string(JSON extra_count LENGTH "${grammar}" extras)

add_library(${PROJECT_NAME} MODULE "${parser_dir}/parser.c")
target_include_directories(${PROJECT_NAME} PRIVATE "${parser_dir}" "${NODE_MODULES}")
target_compile_features(${PROJECT_NAME} PRIVATE c_std_11)
set(output_dir "$<PATH:NORMAL_PATH,${library_dir}>")
set_target_properties(${PROJECT_NAME} PROPERTIES PREFIX "" SUFFIX ".so" OUTPUT_NAME "${language}" C_VISIBILITY_PRESET hidden LIBRARY_OUTPUT_DIRECTORY "${output_dir}" RUNTIME_OUTPUT_DIRECTORY "${output_dir}")
get_target_property(library_suffix ${PROJECT_NAME} SUFFIX)
set(library "${library_dir}/${language}${library_suffix}")
set(scanner_options "$<$<C_COMPILER_ID:GNU>:-fno-strict-aliasing>" ${SCANNER_COMPILE_OPTIONS})

if(external_count GREATER 0)
    set(scanners "")
    set(component_grammars "")
    set(component_names "")
    foreach(component IN LISTS COMPONENTS)
        string(REPLACE "|" ";" component "${component}")
        list(GET component 0 scanner)
        list(GET component 1 component_grammar)
        file(REAL_PATH "${scanner}" scanner)
        file(REAL_PATH "${component_grammar}" component_grammar)
        file(READ "${component_grammar}" component_json)
        string(JSON component_name GET "${component_json}" name)
        list(APPEND scanners "${scanner}")
        list(APPEND component_grammars "${component_grammar}")
        list(APPEND component_names "${component_name}")
    endforeach()
    list(FIND component_names "${name}" extended)
    if(extended GREATER -1)
        list(GET scanners ${extended} scanner)
        target_sources(${PROJECT_NAME} PRIVATE "${scanner}")
        set_property(SOURCE "${scanner}" PROPERTY COMPILE_OPTIONS ${scanner_options})
    else()
        set(external_names "")
        set(extra_flags "")
        math(EXPR external_last "${external_count} - 1")
        math(EXPR extra_last "${extra_count} - 1")
        foreach(index RANGE ${external_last})
            string(JSON external GET "${grammar}" externals ${index})
            string(JSON external_type GET "${external}" type)
            if(external_type STREQUAL "SYMBOL")
                string(JSON external_name GET "${external}" name)
                list(APPEND external_names "${external_name}")
            else()
                list(APPEND external_names "-")
            endif()
            set(extra_flag false)
            if(extra_count GREATER 0)
                foreach(extra_index RANGE ${extra_last})
                    string(JSON extra GET "${grammar}" extras ${extra_index})
                    string(JSON same EQUAL "${external}" "${extra}")
                    if(same)
                        set(extra_flag true)
                    endif()
                endforeach()
            endif()
            list(APPEND extra_flags ${extra_flag})
        endforeach()
        set(base_line "")
        set(host_lines "")
        foreach(scanner component_grammar component_name IN ZIP_LISTS scanners component_grammars component_names)
            file(READ "${component_grammar}" component_json)
            string(JSON component_count LENGTH "${component_json}" externals)
            set(lead -1)
            set(lead_name "")
            math(EXPR component_last "${component_count} - 1")
            foreach(index RANGE ${component_last})
                string(JSON external_type GET "${component_json}" externals ${index} type)
                if(lead EQUAL -1 AND external_type STREQUAL "SYMBOL")
                    set(lead ${index})
                    string(JSON lead_name GET "${component_json}" externals ${index} name)
                endif()
            endforeach()
            string(SUBSTRING "${lead_name}" 0 1 lead_initial)
            if(lead_initial STREQUAL "_")
                set(host_name "_${component_name}${lead_name}")
            else()
                set(host_name "${component_name}_${lead_name}")
            endif()
            list(FIND external_names "${host_name}" host_index)
            list(FIND external_names "${lead_name}" base_index)
            if(host_index GREATER -1)
                math(EXPR offset "${host_index} - ${lead}")
                string(APPEND host_lines "    COMPONENT(${component_name}, ${offset}, ${component_count}) \\\n")
            else()
                math(EXPR offset "${base_index} - ${lead}")
                set(base_line "    COMPONENT(${component_name}, ${offset}, ${component_count}) \\\n")
            endif()
            cmake_path(IS_PREFIX NODE_MODULES "${scanner}" NORMALIZE upstream)
            if(upstream)
                foreach(function IN ITEMS create destroy serialize deserialize scan)
                    set_property(SOURCE "${scanner}" APPEND PROPERTY COMPILE_DEFINITIONS "tree_sitter_${component_name}_external_scanner_${function}=${component_name}_component_${function}")
                endforeach()
                set_property(SOURCE "${scanner}" PROPERTY COMPILE_OPTIONS ${scanner_options})
            else()
                set_property(SOURCE "${scanner}" PROPERTY COMPILE_OPTIONS "-DSCANNER(fn)=${component_name}_component_##fn" ${scanner_options})
            endif()
            target_sources(${PROJECT_NAME} PRIVATE "${scanner}")
        endforeach()
        list(JOIN extra_flags ", " extra_flags)
        file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/components.h" CONTENT "#define COMPONENTS \\\n${base_line}${host_lines}\nstatic const bool EXTRA[] = {${extra_flags}};\n")
        target_sources(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/compose.c")
        set_property(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/compose.c" PROPERTY COMPILE_DEFINITIONS "COMPOSITE=${name}")
        target_include_directories(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    endif()
endif()

string(JSON test_config SET [[{"grammars": [{}], "metadata": {"version": "0.0.0"}}]] grammars 0 name "\"${name}\"")
string(JSON test_config SET "${test_config}" grammars 0 scope "\"source.${name}\"")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/tree-sitter.json" CONTENT "${test_config}\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
set(test_dir "${grammar_dir}/test")
cmake_path(RELATIVE_PATH test_dir BASE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" OUTPUT_VARIABLE test_link)
cmake_path(RELATIVE_PATH library BASE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib" OUTPUT_VARIABLE library_link)
file(CREATE_LINK "${test_link}" "${CMAKE_CURRENT_BINARY_DIR}/test" SYMBOLIC)
file(CREATE_LINK "${library_link}" "${CMAKE_CURRENT_BINARY_DIR}/lib/${name}${CMAKE_SHARED_LIBRARY_SUFFIX}" SYMBOLIC)

if(AST_GREP_CONFIG)
    file(REAL_PATH "${AST_GREP_CONFIG}" AST_GREP_CONFIG)
    cmake_path(GET AST_GREP_CONFIG PARENT_PATH config_dir)
    cmake_path(RELATIVE_PATH library BASE_DIRECTORY "${config_dir}" OUTPUT_VARIABLE project_library)
    cmake_path(RELATIVE_PATH config_dir BASE_DIRECTORY "${library_dir}" OUTPUT_VARIABLE project_root)
    execute_process(
        COMMAND yq -o=json
            [[{"ruleDirs": [], "customLanguages": (.customLanguages | with_entries(select(.value.libraryPath == strenv(AST_GREP_PROJECT_LIBRARY))))}
              | (.customLanguages[].libraryPath) = strenv(AST_GREP_LIBRARY)
              | (.customLanguages[][] | select(key == "outlineRules")) |= strenv(AST_GREP_PROJECT_ROOT) + "/" + .]]
            "${AST_GREP_CONFIG}"
        ENVIRONMENT "AST_GREP_PROJECT_LIBRARY=${project_library}" "AST_GREP_LIBRARY=${language}${library_suffix}" "AST_GREP_PROJECT_ROOT=${project_root}"
        OUTPUT_FILE "${CMAKE_CURRENT_BINARY_DIR}/sgconfig.yml"
        COMMAND_ERROR_IS_FATAL ANY
    )
    file(RENAME "${CMAKE_CURRENT_BINARY_DIR}/sgconfig.yml" "${CMAKE_CURRENT_BINARY_DIR}.yml")
endif()
