using System;
using LanguageExt;
using LanguageExt.Common;
using LanguageExt.Traits;
using static LanguageExt.Prelude;

internal static class FailureChecks {
    internal static Func<Error, bool> Predicate() => static _ => false;
    internal static Func<Error, K<IO, int>> Recovery() => static _ => IO.pure(0);
    internal static IO<int> Failed(Error failure) => IO.fail<int>(failure) | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> Lifted(Fin<int> result) => IO.lift(result) | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> Either(Either<Error, int> result) => IO.lift(result) | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> Mapped(Fin<int> result) => IO.lift(result).Map(static value => value + 1) | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> Named(Error failure) => IO.fail<int>(failure) | @catch(Fail: static _ => IO.pure(0), predicate: static _ => false);
    internal static IO<int> Chained(Error failure) => (IO.fail<int>(failure) | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0))).Map(static value => value + 1);
    internal static IO<int> Branch(bool failed, Error failure) => (failed ? IO.fail<int>(failure) : IO.pure(1)) | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> Switched(bool failed, Error failure) => (failed switch { true => IO.fail<int>(failure), false => IO.pure(1) }) | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> Unknown(IO<int> supplied) => supplied | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> Ninth(int one, int two, int three, int four, int five, int six, int seven, int eight, K<IO, int> supplied) => supplied.As() | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> Extended(this K<IO, int> supplied, int ignored) => supplied.As() | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    internal static IO<int> MethodGroup(IO<int> supplied, Func<Error, bool> predicate, Func<Error, K<IO, int>> recovery) => supplied | @catch(predicate: predicate, Fail: recovery);
    internal static IO<int> EvaluatedArguments(Error failure) => IO.fail<int>(failure) | @catch(Fail: Recovery(), predicate: Predicate());
    internal static IO<int> Mutated(Error failure) {
        IO<int> supplied = IO.pure(1);
        supplied = IO.fail<int>(failure);
        return supplied | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    }
    internal static IO<int> Nested(Error failure) => IO.fail<int>(failure) | @catch(predicate: static _ => false, Fail: static error => IO.fail<int>(error) | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0)));
    internal static IO<int> Called(Error failure) => Ninth(0, 0, 0, 0, 0, 0, 0, 0, IO.fail<int>(failure));
    internal static IO<int> Reordered(Error failure) => Ninth(supplied: IO.fail<int>(failure), eight: 0, seven: 0, six: 0, five: 0, four: 0, three: 0, two: 0, one: 0);
    internal static IO<int> Reduced(Error failure) => IO.fail<int>(failure).Extended(0);
}

internal static class DeferredChecks {
    internal static IO<int> Pure() => IO.pure(1).Catch(static _ => false, static _ => IO.pure(0));
    internal static IO<int> Lifted() => IO.lift(static () => 1).Catch(static _ => false, static _ => IO.pure(0));
    internal static IO<int> Mapped() => IO.lift(static () => 1).Map(static value => value + 1).Catch(static _ => false, static _ => IO.pure(0));
    internal static IO<int> Always(Error failure) => IO.fail<int>(failure).Catch(static _ => true, static _ => IO.pure(0));
    internal static IO<int> Local() {
        IO<int> supplied = IO.lift(static () => 1);
        return supplied.Catch(static _ => false, static _ => IO.pure(0));
    }
}

internal static class DeferredWrappers {
    internal static IO<int> Uninterruptible(Error failure) => IO.fail<int>(failure).Uninterruptible().Catch(static _ => false, static _ => IO.pure(0));
    internal static IO<int> Timed(Error failure) => IO.fail<int>(failure).Timeout(TimeSpan.FromSeconds(1)).Catch(static _ => false, static _ => IO.pure(0));
    internal static IO<int> Converted(Error failure) {
        IO<int> supplied = failure;
        return supplied.Catch(static _ => false, static _ => IO.pure(0));
    }
    internal static IO<int> Recovered(Error failure) => IO.fail<int>(failure).Catch(static _ => true, static _ => IO.pure(0)).Catch(static _ => false, static _ => IO.pure(1));
}

internal static class ReachingFailures {
    internal static IO<int> LaterWrite(Error failure) {
        IO<int> supplied = IO.pure(1);
        IO<int> first = supplied.Catch(static _ => false, static _ => IO.pure(0));
        supplied = IO.fail<int>(failure);
        return first;
    }
    internal static IO<int> Disjoint(Error failure, bool failed) {
        IO<int> supplied = IO.pure(1);
        if (failed) {
            supplied = IO.fail<int>(failure);
            return supplied;
        }
        return supplied.Catch(static _ => false, static _ => IO.pure(0));
    }
    internal static IO<int> Looped(Error failure, bool failed) {
        IO<int> supplied = IO.pure(1);
        while (failed) {
            supplied = IO.fail<int>(failure);
            failed = false;
        }
        return supplied | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    }
    internal static IO<int> FinallyPure(Error failure) {
        IO<int> supplied = IO.pure(1);
        try { supplied = IO.fail<int>(failure); }
        finally { supplied = IO.pure(1); }
        return supplied.Catch(static _ => false, static _ => IO.pure(0));
    }
    internal static IO<int> FinallyFailed(Error failure) {
        IO<int> supplied = IO.pure(1);
        try { supplied = IO.pure(1); }
        finally { supplied = IO.fail<int>(failure); }
        return supplied | @catch(predicate: static _ => false, Fail: static _ => IO.pure(0));
    }
}
