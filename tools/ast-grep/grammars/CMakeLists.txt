cmake_minimum_required(VERSION 4.4)

# --- [OPERATIONS] -----------------------------------------------------------------------

function(spans grammar rule out)
    string(JSON members GET "${grammar}" rules ${rule} members)
    string(JSON count LENGTH "${members}")
    math(EXPR last "${count} - 1")
    set(rows "")
    foreach(index RANGE 0 ${last} 2)
        math(EXPR maximum_at "${index} + 1")
        string(JSON minimum GET "${members}" ${index} value)
        string(JSON maximum GET "${members}" ${maximum_at} value)
        string(APPEND rows "{${minimum}, ${maximum}}, ")
    endforeach()
    set(${out} "${rows}" PARENT_SCOPE)
endfunction()

# --- [COMPOSITION] ----------------------------------------------------------------------

project(grammar LANGUAGES C)

set(GRAMMAR_SOURCE "" CACHE FILEPATH "Grammar extension or composition")
set(NODE_MODULES "" CACHE PATH "Installed packages")
set(SCANNER "" CACHE STRING "External scanner")
set(SCANNER_COMPILE_OPTIONS "" CACHE STRING "External scanner compiler options")

file(REAL_PATH "${GRAMMAR_SOURCE}" GRAMMAR_SOURCE)
file(REAL_PATH "${NODE_MODULES}" NODE_MODULES)
cmake_path(GET CMAKE_CURRENT_BINARY_DIR PARENT_PATH published_dir)
cmake_path(GET published_dir FILENAME language)
cmake_path(GET published_dir PARENT_PATH library_dir)
set(library_suffix ".so")
set(declared_dir "${CMAKE_CURRENT_BINARY_DIR}/declared")

execute_process(COMMAND tree-sitter generate --no-parser --output "${declared_dir}" "${GRAMMAR_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND tree-sitter --version OUTPUT_VARIABLE generator COMMAND_ERROR_IS_FATAL ANY)
file(READ "${declared_dir}/grammar.json" grammar)
string(JSON name GET "${grammar}" name)
string(SHA256 parser_key "${generator}${grammar}")
set(parser_dir "${published_dir}/parser/${parser_key}")
file(MAKE_DIRECTORY "${published_dir}/src")

if(NOT EXISTS "${parser_dir}/parser.c")
    execute_process(COMMAND tree-sitter generate --output "${CMAKE_CURRENT_BINARY_DIR}/parser" "${declared_dir}/grammar.json" COMMAND_ERROR_IS_FATAL ANY)
    file(GLOB superseded LIST_DIRECTORIES true "${published_dir}/parser/*")
    file(REMOVE_RECURSE ${superseded})
    file(MAKE_DIRECTORY "${published_dir}/parser")
    file(RENAME "${CMAKE_CURRENT_BINARY_DIR}/parser" "${parser_dir}" RESULT published)
endif()

file(SHA256 "${CMAKE_CURRENT_LIST_FILE}" build_key)
set(library_key "${parser_key}${build_key}${CMAKE_C_COMPILER_ID}${CMAKE_C_COMPILER_VERSION}${SCANNER_COMPILE_OPTIONS}")
string(JSON hole_forms ERROR_VARIABLE plain GET "${grammar}" rules _hole_forms members)
string(JSON word_start ERROR_VARIABLE wordless GET "${grammar}" rules _word_start)
string(JSON space ERROR_VARIABLE spaceless GET "${grammar}" rules _space)
string(JSON constants ERROR_VARIABLE constantless GET "${grammar}" rules _constants members)

if(NOT wordless)
    spans("${grammar}" _word_start start_rows)
    spans("${grammar}" _word_character character_rows)
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/word.h" CONTENT "#define WORD_START ${start_rows}\n#define WORD_CHARACTER ${character_rows}\n")
endif()
if(NOT spaceless)
    spans("${grammar}" _space space_rows)
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/space.h" CONTENT "#define SPACE ${space_rows}\n")
endif()
if(NOT constantless)
    string(JSON constant_count LENGTH "${constants}")
    math(EXPR constant_last "${constant_count} - 1")
    set(constant_rows "")
    foreach(index RANGE 0 ${constant_last} 2)
        math(EXPR value_at "${index} + 1")
        string(JSON constant GET "${constants}" ${index} value)
        string(JSON value GET "${constants}" ${value_at} value)
        string(APPEND constant_rows "#define ${constant} ${value}\n")
    endforeach()
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/constants.h" CONTENT "${constant_rows}")
endif()
if(SCANNER)
    file(REAL_PATH "${SCANNER}" SCANNER)
    execute_process(COMMAND "${CMAKE_C_COMPILER}" -E "-I${parser_dir}" "-I${NODE_MODULES}" "-I${CMAKE_CURRENT_BINARY_DIR}" "${SCANNER}" OUTPUT_VARIABLE scanned COMMAND_ERROR_IS_FATAL ANY)
    string(APPEND library_key "${scanned}")
endif()
if(NOT plain)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/compose.c" compose_key)
    string(APPEND library_key "${compose_key}")
endif()
string(SHA256 library_key "${library_key}")
set(stored_library "${parser_dir}/${library_key}${library_suffix}")

if(NOT EXISTS "${stored_library}")
    file(GLOB superseded "${parser_dir}/*${library_suffix}")
    file(REMOVE ${superseded})
    add_library(${PROJECT_NAME} MODULE "${parser_dir}/parser.c")
    target_include_directories(${PROJECT_NAME} PRIVATE "${parser_dir}" "${NODE_MODULES}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_features(${PROJECT_NAME} PRIVATE c_std_11)
    set_target_properties(${PROJECT_NAME} PROPERTIES PREFIX "" SUFFIX "${library_suffix}" OUTPUT_NAME "${language}" C_VISIBILITY_PRESET hidden)
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E rename "$<TARGET_FILE:${PROJECT_NAME}>" "${stored_library}" VERBATIM)
    set(scanner_options "$<$<C_COMPILER_ID:GNU>:-fno-strict-aliasing>" ${SCANNER_COMPILE_OPTIONS})

    if(SCANNER)
        target_sources(${PROJECT_NAME} PRIVATE "${SCANNER}")
        set_property(SOURCE "${SCANNER}" PROPERTY COMPILE_OPTIONS ${scanner_options})
    endif()

    if(NOT plain)
        string(JSON external_count LENGTH "${grammar}" externals)
        math(EXPR hole_symbol "${external_count} - 1")
        string(JSON form_count LENGTH "${hole_forms}")
        math(EXPR form_last "${form_count} - 1")
        set(form_rows "")
        foreach(index RANGE 0 ${form_last} 2)
            math(EXPR closing_at "${index} + 1")
            string(JSON opening GET "${hole_forms}" ${index} value)
            string(JSON closing GET "${hole_forms}" ${closing_at} value)
            string(APPEND form_rows "{\"${opening}\", '${closing}'}, ")
        endforeach()
        spans("${grammar}" _hole_tail tail_rows)
        string(JSON hole_words GET "${grammar}" rules _hole_words members)
        string(JSON word_count LENGTH "${hole_words}")
        math(EXPR word_last "${word_count} - 1")
        set(word_rows "")
        set(word_size 1)
        foreach(index RANGE 0 ${word_last})
            string(JSON word GET "${hole_words}" ${index} value)
            string(LENGTH "${word}" word_length)
            if(word_length GREATER_EQUAL word_size)
                math(EXPR word_size "${word_length} + 1")
            endif()
            string(APPEND word_rows "\"${word}\", ")
        endforeach()
        file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/holes.h" CONTENT "#define COMPOSITE_SCAN tree_sitter_${name}_external_scanner_scan\n#define HOLE_SYMBOL ${hole_symbol}\n#define HOLE_FORMS ${form_rows}\n#define HOLE_TAIL ${tail_rows}\n#define HOLE_WORDS ${word_rows}\n#define HOLE_WORD_SIZE ${word_size}\n")
        string(JSON dialect GET "${grammar}" inherits)
        foreach(function IN ITEMS create destroy serialize deserialize)
            set_property(SOURCE "${SCANNER}" APPEND PROPERTY COMPILE_DEFINITIONS "tree_sitter_${dialect}_external_scanner_${function}=tree_sitter_${name}_external_scanner_${function}")
        endforeach()
        set_property(SOURCE "${SCANNER}" APPEND PROPERTY COMPILE_DEFINITIONS "tree_sitter_${dialect}_external_scanner_scan=dialect_scan")
        target_sources(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/compose.c")
    endif()
endif()

add_custom_target(publish ALL
    COMMAND "${CMAKE_COMMAND}" -E rename "${declared_dir}/grammar.json" "${published_dir}/src/grammar.json"
    COMMAND "${CMAKE_COMMAND}" -E rename "${declared_dir}/node-types.json" "${published_dir}/src/node-types.json"
    COMMAND "${CMAKE_COMMAND}" -E copy "${stored_library}" "${CMAKE_CURRENT_BINARY_DIR}/published${library_suffix}"
    COMMAND "${CMAKE_COMMAND}" -E rename "${CMAKE_CURRENT_BINARY_DIR}/published${library_suffix}" "${library_dir}/${language}${library_suffix}"
    VERBATIM
)
if(TARGET ${PROJECT_NAME})
    add_dependencies(publish ${PROJECT_NAME})
endif()
