class FaultRecovery(Struct, frozen=True, gc=False):
    # producer oneof: at most one of terminal/transient/retry_after is set per fault, exactly as `GeometryPayload`
    # carries its three arms. The two `Empty` arms are presence-only verdicts and the third carries the window itself.
    # BOTH directions of the `Recovery` correspondence seat on this shape, so a forward and an inverse that drift into
    # two truths have nowhere to live. `reliability/faults#FAULT` owns the interior verdict vocabulary whole and this
    # page owns the cell alone: wire absence is NOT a fourth interior case but the `Option` the INGRESS fold answers,
    # so a frame minted before the field existed states that it carried no verdict without every interior consumer
    # matching a state no interior value can hold. Egress takes a bare verdict for the same reason — unstated is what
    # a legacy peer sends, never a posture this branch adopts. Defaulting an unread verdict to `terminal` is what the
    # pair forecloses: it strands every legacy peer's re-drivable fault as permanent, the drift this slot exists for.
    terminal: Empty | None = None
    transient: Empty | None = None
    retry_after: Span | None = None

    @staticmethod
    def of(recovery: Recovery) -> "FaultRecovery":
        # ONE total egress fold, and TOTAL on a bare verdict rather than on an `Option`: unstated is an INGRESS fact a
        # legacy peer's frame carries, never a posture this branch may adopt, so minting a silent cell is unspellable
        # here and every frame this host writes states its verdict. Nine fractional digits is the Duration JSON
        # ceiling, so this render loses no window `stated` ever admits; `json_format` re-normalizes an emitted spelling
        # onto its own 0/3/6/9-digit grid, so a round trip preserves every VALUE while settling its SPELLING at that
        # grid — which is why `stated` parses a window rather than comparing one.
        match recovery:
            case Recovery(tag="terminal"):
                return FaultRecovery(terminal=Empty())
            case Recovery(tag="transient"):
                return FaultRecovery(transient=Empty())
            case Recovery(tag="throttled", throttled=seconds):
                return FaultRecovery(retry_after=f"{seconds:.9f}s")
            case _ as unreachable:
                assert_never(unreachable)

    @staticmethod
    def stated(cell: "FaultRecovery | None") -> RuntimeRail[Option[Recovery]]:
        # ONE total ingress fold: an absent cell and an unset one both answer `Nothing`, one set arm answers its case,
        # and TWO set arms refuse. At-most-one-arm-set is a property of the VALUE, so this fold is where the oneof law
        # holds — a boot census reads declarations and proves nothing about it, exactly as it proves nothing about
        # `GeometryPayload`. `Span` constrains the window at the CODEC and a Python annotation constrains nothing at
        # construction, so the guard rides AHEAD of the parse and the `float` is total on every value that reaches it;
        # a malformed window refuses rather than folding to `terminal`, so a producer's broken cell can never read
        # downstream as a producer's terminal verdict. A zero window stays `throttled` and never coalesces onto
        # `transient`: the fault tier keeps retry-immediately separable from a dependency that answered nothing.
        match cell:
            case None | FaultRecovery(terminal=None, transient=None, retry_after=None):
                return Ok(Nothing)
            case FaultRecovery(terminal=Empty(), transient=None, retry_after=None):
                return Ok(Some(Recovery(terminal=None)))
            case FaultRecovery(terminal=None, transient=Empty(), retry_after=None):
                return Ok(Some(Recovery(transient=None)))
            case FaultRecovery(terminal=None, transient=None, retry_after=str() as window):
                return (
                    Ok(Some(Recovery(throttled=float(window.removesuffix("s")))))
                    if _WINDOW.fullmatch(window)
                    else Error(SHAPES_WINDOW.raised(window))
                )
            case _:
                return Error(SHAPES_ARMS.raised())

class Admission(Struct, frozen=True, gc=False):
    # ONE correspondence row: the `msgspec.inspect` node classes a slot may hold for its proto coordinate, and the one
    # narrowing the coordinate DEMANDS rather than merely admits. `floor` rides the unsigned 64-bit rows alone, where
    # proto3 JSON renders a decimal STRING that `convert(strict=False)` coerces and a signed spelling there decodes a
    # negative no producer can have written; an EMPTY `nodes` tuple is a declared refusal, admitting no slot at all.
    nodes: tuple[type[Type], ...]
    floor: bool = False


class Closure(Struct, frozen=True, gc=False):
    # ONE registration of a closed family against the row table keyed on it. The family and the table sit ABOVE this
    # module on the import rail, so the composition root that already runs the gate hands both down — reaching upward
    # from here for `BINDINGS`, `CLASSIFICATION_ROWS`, `FORMAT_ROWS`, `_FAULT_STATUS`, or `POLICY` inverts the rail
    # this module roots. A DELIBERATELY partial table registers NO row: `CIRCUIT` and `RATES` declare their own
    # absence as the policy, so folding them in here would refuse the process their landed partiality exists to serve.
    table: str
    members: frozenset[str]
    rostered: frozenset[str]

    @staticmethod
    def of(table: str, family: "type[StrEnum] | tuple[str, ...]", rostered: Map[str, object]) -> "Closure":
        # ONE seat for both closed-family spellings the branch mints — a `StrEnum` iterates its own members and a
        # `Literal` family arrives as its `get_args` tuple, each a set of `str` either way — so a root registering a
        # table picks no lifting and two registrations can never disagree about what a family's member set is.
        return Closure(table=table, members=frozenset(family), rostered=frozenset(rostered.keys()))

# --- [OPERATIONS] -----------------------------------------------------------------------


def _borne(node: Type) -> Type:
    # `T | None` is the ONE union this vocabulary spells — the explicit-presence slot — so the correspondence reads the
    # borne arm while the presence law reads the lift itself. A union carrying two borne arms is no presence spelling
    # and stays whole, where the row it fails to match is what names it.
    arms = tuple(arm for arm in node.types if not isinstance(arm, NoneType)) if isinstance(node, UnionType) else ()
    return arms[0] if len(arms) == 1 else node


def _element(wire: FieldDescriptor, node: Type) -> tuple[str, FieldDescriptor, Type]:
    # cardinality unwrap, licensed by the DESCRIPTOR's own columns rather than by a slot's shape: a map field carries
    # its own generated entry message, so the coordinate the value is proved against is that entry's `value` field and
    # never the entry message itself, and proto3 renders every map key as a string the slot's `dict[str, ...]` typing
    # already fixes. A container the slot does not spell returns its own name for the census, so `repeated string`
    # against a scalar slot names the cardinality rather than passing on a type comparison that was never reached.
    mapped = wire.message_type is not None and wire.message_type.GetOptions().map_entry
    match ("map" if mapped else "repeated" if wire.is_repeated else "", node):
        case ("map", DictType(value_type=value)):
            return "", wire.message_type.fields_by_name["value"], value
        case ("repeated", ListType(item_type=item) | VarTupleType(item_type=item)):
            return "", wire, item
        case (container, borne):
            return container, wire, borne


def _seated(subject: str, key: str, row: Admission, node: Type) -> Block[str]:
    return (
        Block.empty() if isinstance(node, row.nodes) else Block.singleton(f"{subject}:slot-{type(node).__name__}-against-{key}")
    ).append(
        Block.empty()
        if not row.floor or (isinstance(node, IntType) and node.ge == 0)
        else Block.singleton(f"{subject}:64-bit-slot-without-WireU64-floor")
    )


def _corresponds(name: str, wire: FieldDescriptor, field: Field) -> Block[str]:
    # completeness a name-set census cannot reach: a `repeated string` mapped onto a scalar `str`, a message mapped
    # onto a `str`, and a `bool` mapped onto an `int` all hold identical NAMES, and every one of them decodes as a
    # `ValidationError` at a customer's first RPC. Reading the descriptor's type, cardinality, and presence columns
    # against the slot's own `msgspec.inspect` node moves each of those to the boot this gate already runs.
    subject, borne = f"{name}.{wire.name}", _borne(field.type)
    presence = (
        Block.empty()
        if borne is field.type or wire.has_presence
        else Block.singleton(f"{subject}:optional-slot-without-producer-presence")
    )
    container, coordinate, node = _element(wire, borne)
    named = coordinate.message_type.full_name if coordinate.message_type is not None else ""
    key = named if named in _CORRESPONDENCE else f"type-{coordinate.type}"
    return presence.append(
        Block.singleton(f"{subject}:cardinality-{container}-against-{type(node).__name__}")
        if container
        else _CORRESPONDENCE.try_find(key)
        .map(lambda row: _seated(subject, key, row, node))
        .default_value(Block.singleton(f"{subject}:coordinate-{key}-unrostered"))
    )


def _drift(name: str, struct: type[Struct], message: type[Message]) -> Block[str]:
    # importing channels_pb2 registered its serialized file in the default pool — this lookup is a read, never AddSerializedFile.
    compiled = descriptor_pool.Default().FindMessageTypeByName(message.DESCRIPTOR.full_name)
    slots = {field.encode_name: field for field in type_info(struct).fields}
    absent = Block.of_seq(sorted(frozenset(compiled.fields_by_name) - frozenset(slots))).map(lambda gap: f"{name}.{gap}:producer-field-unmapped")
    phantom = Block.of_seq(sorted(frozenset(slots) - frozenset(compiled.fields_by_name))).map(lambda gap: f"{name}.{gap}:slot-never-decodes")
    corresponds = Block.of_seq(wire for wire in compiled.fields_by_name.values() if wire.name in slots).collect(
        lambda wire: _corresponds(name, wire, slots[wire.name])
    )
    return absent.append(phantom).append(corresponds)


def _closed(row: Closure) -> Block[str]:
    # closure coverage, the third census this gate owes: a table keyed on a closed family answers a `try_find` absence
    # or a defaulted status for an unrostered member, so the member that was never rowed surfaces at a caller's first
    # dial rather than at boot. Both directions read, exactly as the drift and mirror arms read theirs — a rostered key
    # outside the family is the same stale cell from the other side, which an enum-keyed table forecloses statically
    # and a runtime-built one does not.
    unrostered = Block.of_seq(sorted(row.members - row.rostered)).map(lambda gap: f"{row.table}.{gap}:closure-member-unrostered")
    stale = Block.of_seq(sorted(row.rostered - row.members)).map(lambda gap: f"{row.table}.{gap}:row-outside-closure")
    return unrostered.append(stale)


def _mirror(name: str, struct: type[Struct], roster: tuple[str, ...]) -> Block[str]:
    # the mirror census: a positional record carries no names on the wire, so the whole contract is that this struct
    # holds exactly the roster's slots and holds them in exactly the roster's order. An empty roster proves nothing
    # against itself by construction, which is the point — it DECLARES declaration order as the key order, and the
    # spelled roster below is what a producer's out-of-order append turns into a checkable claim.
    slots = tuple(field.name for field in type_info(struct).fields)
    order = MIRROR_ORDER[name]
    absent = Block.of_seq(sorted(frozenset(slots) - frozenset(order))).map(lambda gap: f"{name}.{gap}:slot-outside-key-roster")
    phantom = Block.of_seq(sorted(frozenset(order) - frozenset(slots))).map(lambda gap: f"{name}.{gap}:key-without-slot")
    seated = Block.empty() if len(order) == len(slots) else Block.singleton(f"{name}:roster-{len(order)}-against-{len(slots)}-slots")
    # every nested slot this struct declares must key a mirror row, or the decoder hands `convert` a raw positional
    # list where a struct stands and the whole sub-record decodes as a type error one level below the fault subject.
    unrouted = Block.of_seq(sorted(slot for slot in slots if slot in MIRROR_NESTED and MIRROR_NESTED[slot] not in MIRROR_ORDER)).map(
        lambda slot: f"{name}.{slot}:nested-row-unregistered"
    )
    return absent.append(phantom).append(seated).append(unrouted)


def _service(name: WireService, methods: tuple[WireMethod, ...], proof: ServiceProof) -> Block[str]:
    # `FindServiceByName` raises `KeyError` on an unregistered name under both the upb and pure-python pools
    # (`.api/protobuf.md`) — precisely the drift this arm exists to name — so the lookup rides the faults fence under
    # that one class and its refusal lands as a census row rather than as a boot traceback. A BROKERED row
    # short-circuits before the pool is touched: there is nothing compiled for it to miss.
    return (
        Block.empty()
        if proof is ServiceProof.BROKERED
        else boundary(SHAPES_SERVICES, lambda: descriptor_pool.Default().FindServiceByName(name), catch=KeyError)
        .map(
            lambda compiled: Block.of_seq(sorted(frozenset(methods) - frozenset(compiled.methods_by_name))).map(
                lambda gap: f"{name}.{gap}:method-unrostered"
            )
        )
        .default_value(Block.singleton(f"{name}:service-name-unresolvable"))
    )


def aligned(closed: Block[Closure]) -> RuntimeRail[int]:
    # ONE boot answer over all three registries and every closed family a caller registers: a descriptor-backed row
    # proves against its compiled message, a mirror row against its producer's key roster, a service row against the
    # compiled service its dial path names, and a `Closure` row against the table keyed on it. The name spaces stay
    # disjoint EXCEPT where one leaf legitimately crosses both wires — so the duplicate check runs per registry rather
    # than over their union, where the shared `wire_provenance` leaf would read as a collision. The closure roster
    # arrives as an argument because those tables sit above this module on the import rail and only the composition
    # root reaches all of them; it costs a boot read over a few small tables and reads no installed state, exactly the
    # price `transport/serve#SERVE` already fixed for seating this gate ahead of every install.
    rows = len(PROTO_VOCABULARY) + len(MIRROR_VOCABULARY) + len(SERVICE_VOCABULARY) + len(closed)
    doubled = any(
        unique != len(table)
        for unique, table in (
            (len(frozenset(name for name, _, _ in PROTO_VOCABULARY)), PROTO_VOCABULARY),
            (len(MIRROR_ORDER), MIRROR_VOCABULARY),
            (len(frozenset(name for name, _, _ in SERVICE_VOCABULARY)), SERVICE_VOCABULARY),
            (len(frozenset(row.table for row in closed)), closed),
        )
    )
    drift = (
        Block.of_seq(PROTO_VOCABULARY)
        .collect(lambda row: _drift(*row))
        .append(Block.of_seq(MIRROR_VOCABULARY).collect(lambda row: _mirror(*row)))
        .append(Block.of_seq(SERVICE_VOCABULARY).collect(lambda row: _service(*row)))
        .append(closed.collect(_closed))
    )
    return (
        Error(SHAPES_DOUBLED.raised())
        if doubled
        else Ok(rows)
        if drift.is_empty()
        else Error(SHAPES_DRIFT.raised(";".join(drift)))
    )
